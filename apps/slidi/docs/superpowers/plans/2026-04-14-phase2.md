export type Provider = "openai" | "anthropic" | "gemini";

const HISTORY_CAP = 20;
const KEY_PREFIX = "slidi_api_key_";
const PROVIDER_KEY = "slidi_provider";
const THEME_KEY = "slidi_theme";
const HISTORY_KEY = "slidi_history";
const HISTORY_INDEX_KEY = "slidi_history_index";

interface SlidiState {
  // BYOK — per-provider keys
  keys: Record<Provider, string>;
  apiKey: string;
  provider: Provider;
  setApiKey: (key: string, provider: Provider) => void;
  clearApiKey: () => void;

  setTheme: (id: ThemeId) => void;

  // Chat

  addMessage: (msg: ChatMessage) => void;

  generatedCode: string;
  currentVersionId: string;
  pushVersion: (code: string) => void;
  undo: () => void;
  redo: () => void;

  setIsGenerating: (v: boolean) => void;

function loadFromStorage(): {
  keys: Record<Provider, string>;
  apiKey: string;
  provider: Provider;
  theme: ThemeId;
  history: string[];
  historyIndex: number;
  generatedCode: string;
} {
  if (typeof window === "undefined") {
    return {
      keys: { openai: "", anthropic: "", gemini: "" },
      apiKey: "",
      provider: "openai",
      theme: "minimal",
      history: [],
      historyIndex: -1,
      generatedCode: "",
    };
  }

  const provider = (localStorage.getItem(PROVIDER_KEY) as Provider) ?? "openai";
  const keys: Record<Provider, string> = {
    openai: localStorage.getItem(`${KEY_PREFIX}openai`) ?? "",
    anthropic: localStorage.getItem(`${KEY_PREFIX}anthropic`) ?? "",
    gemini: localStorage.getItem(`${KEY_PREFIX}gemini`) ?? "",
  };

  // Migrate legacy single-key storage
  const legacyKey = localStorage.getItem("slidi_api_key");
  if (legacyKey && !keys[provider]) {
    keys[provider] = legacyKey;
    localStorage.setItem(`${KEY_PREFIX}${provider}`, legacyKey);
    localStorage.removeItem("slidi_api_key");
  }

  const theme = (localStorage.getItem(THEME_KEY) as ThemeId) ?? "minimal";

  let history: string[] = [];
  let historyIndex = -1;
  try {
    const stored = localStorage.getItem(HISTORY_KEY);
    if (stored) history = JSON.parse(stored);
    const storedIdx = localStorage.getItem(HISTORY_INDEX_KEY);
    if (storedIdx !== null) historyIndex = parseInt(storedIdx, 10);
  } catch {
    history = [];
    historyIndex = -1;
  }

  const generatedCode = historyIndex >= 0 && history[historyIndex] ? history[historyIndex] : "";

  return { keys, apiKey: keys[provider], provider, theme, history, historyIndex, generatedCode };
}

export const useSlidiStore = create<SlidiState>()((set, get) => ({
  ...loadFromStorage(),
  currentVersionId: "",

  setApiKey: (key, provider) => {
    localStorage.setItem(`${KEY_PREFIX}${provider}`, key);
    localStorage.setItem(PROVIDER_KEY, provider);
    set((state) => ({
      keys: { ...state.keys, [provider]: key },
      apiKey: key,
      provider,
    }));
  },

  clearApiKey: () => {
    (["openai", "anthropic", "gemini"] as Provider[]).forEach((p) => {
      localStorage.removeItem(`${KEY_PREFIX}${p}`);
    });
    localStorage.removeItem(PROVIDER_KEY);
    set({ keys: { openai: "", anthropic: "", gemini: "" }, apiKey: "", provider: "openai" });
  },

  setTheme: (id) => {
    localStorage.setItem(THEME_KEY, id);
    set({ theme: id });
  },

  messages: [],
  addMessage: (msg) => set((state) => ({ messages: [...state.messages, msg] })),
  clearMessages: () => set({ messages: [] }),

  pushVersion: (code) => {
    const { history, historyIndex } = get();
    // Truncate any redo history ahead of current index
    const base = history.slice(0, historyIndex + 1);
    const next = [...base, code].slice(-HISTORY_CAP);
    const nextIndex = next.length - 1;
    localStorage.setItem(HISTORY_KEY, JSON.stringify(next));
    localStorage.setItem(HISTORY_INDEX_KEY, String(nextIndex));
    set({ history: next, historyIndex: nextIndex, generatedCode: code });
  },

  undo: () => {
    const { history, historyIndex } = get();
    if (historyIndex <= 0) return;
    const nextIndex = historyIndex - 1;
    localStorage.setItem(HISTORY_INDEX_KEY, String(nextIndex));
    set({ historyIndex: nextIndex, generatedCode: history[nextIndex] });
  },

  redo: () => {
    const { history, historyIndex } = get();
    if (historyIndex >= history.length - 1) return;
    const nextIndex = historyIndex + 1;
    localStorage.setItem(HISTORY_INDEX_KEY, String(nextIndex));
    set({ historyIndex: nextIndex, generatedCode: history[nextIndex] });
  },

  setCurrentVersionId: (id) => set({ currentVersionId: id }),

  isGenerating: false,
  setIsGenerating: (v) => set({ isGenerating: v }),
}));
```

import { buildPrompt } from "./prompt";
import { ChatMessage } from "@/store/slidiStore";

type Provider = "openai" | "anthropic" | "gemini";

function getProvider(apiKey: string): Provider {
  if (apiKey.startsWith("sk-ant-")) return "anthropic";
  if (apiKey.startsWith("AIza")) return "gemini";
  return "openai";
}

function extractCode(raw: string): string {
  const fenced = raw.match(/```(?:jsx|tsx|js|ts|javascript)?\n([\s\S]*?)```/);
  if (fenced) return fenced[1].trim();
  return raw.trim();
}

async function callOpenAI(
  messages: ChatMessage[],
  apiKey: string,
  systemPrompt: string
): Promise<string> {
  const body = {
    model: "gpt-4o",
    messages: [
      { role: "system", content: systemPrompt },
      ...messages.map((m) => ({
        role: m.role === "user" ? "user" : "assistant",
        content: m.content,
      })),
    ],
    max_tokens: 4096,
    temperature: 0.7,
  };

  const res = await fetch("https://api.openai.com/v1/chat/completions", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${apiKey}`,
    },
    body: JSON.stringify(body),
  });

  if (!res.ok) {
    const err = await res.json().catch(() => ({}));
    throw new Error(err?.error?.message ?? `OpenAI error ${res.status}`);
  }

  const data = await res.json();
  return extractCode(data.choices[0].message.content as string);
}

async function callAnthropic(
  messages: ChatMessage[],
  apiKey: string,
  systemPrompt: string
): Promise<string> {
  const anthropicMessages = messages
    .filter(
      (m) =>
        m.role === "user" ||
        (m.role === "system" && !m.isOutput && !m.isError && m.content !== "Generating...")
    )
    .map((m) => ({
      role: m.role === "user" ? "user" : "assistant",
      content: m.content,
    }));

  const body = {
    model: "claude-sonnet-4-6",
    max_tokens: 4096,
    system: systemPrompt,
    messages: anthropicMessages,
  };

  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "x-api-key": apiKey,
      "anthropic-version": "2023-06-01",
      "anthropic-dangerous-direct-browser-access": "true",
    },
    body: JSON.stringify(body),
  });

  if (!res.ok) {
    const err = await res.json().catch(() => ({}));
    throw new Error(err?.error?.message ?? `Anthropic error ${res.status}`);
  }

  const data = await res.json();
  return extractCode(data.content[0].text as string);
}

async function callGemini(
  messages: ChatMessage[],
  apiKey: string,
  systemPrompt: string
): Promise<string> {
  const userMessages = messages.filter((m) => m.role === "user");
  const lastUser = userMessages[userMessages.length - 1]?.content ?? "";

  const body = {
    contents: [
      {
        role: "user",
        parts: [{ text: `${systemPrompt}\n\n---\n\nUser request: ${lastUser}` }],
      },
    ],
    generationConfig: {
      maxOutputTokens: 4096,
      temperature: 0.7,
    },
  };

  const res = await fetch(
    `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`,
    {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(body),
    }
  );

  if (!res.ok) {
    const err = await res.json().catch(() => ({}));
    throw new Error(err?.error?.message ?? `Gemini error ${res.status}`);
  }

  const data = await res.json();
  const text = data.candidates?.[0]?.content?.parts?.[0]?.text as string;
  return extractCode(text);
}

export async function generatePresentation(
  messages: ChatMessage[],
  apiKey: string,
  themeBlock: string
): Promise<string> {
  if (!apiKey) throw new Error("No API key configured.");

  const systemPrompt = buildPrompt(themeBlock);
  const provider = getProvider(apiKey);

  switch (provider) {
    case "openai":
      return callOpenAI(messages, apiKey, systemPrompt);
    case "anthropic":
      return callAnthropic(messages, apiKey, systemPrompt);
    case "gemini":
      return callGemini(messages, apiKey, systemPrompt);
  }
}

export const runtime = "edge";

export async function POST(request: Request) {
  const env = process.env as unknown as CloudflareEnv;

  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return Response.json({ error: "Invalid JSON" }, { status: 400 });
  }

  if (
    typeof body !== "object" ||
    body === null ||
    typeof (body as Record<string, unknown>).code !== "string"
  ) {
    return Response.json({ error: "Missing required field: code" }, { status: 400 });
  }

  const code = ((body as Record<string, unknown>).code as string).trim();
  if (!code) {
    return Response.json({ error: "code must not be empty" }, { status: 400 });
  }
  if (code.length > 512_000) {
    return Response.json({ error: "code exceeds 500 KB limit" }, { status: 400 });
  }

  const id = crypto.randomUUID();

  try {
    await env.DB.prepare(
      "INSERT INTO presentations (id, code_content) VALUES (?1, ?2)"
    )
      .bind(id, code)
      .run();
  } catch (err) {
    console.error("D1 insert failed:", err);
    return Response.json({ error: "Database error" }, { status: 500 });
  }

  return Response.json({ id, url: `/view/${id}` });
}
```

export const runtime = "edge";

export async function GET(
  _request: Request,
  { params }: { params: { id: string } }
) {
  const env = process.env as unknown as CloudflareEnv;
  const { id } = params;

  if (!id || typeof id !== "string") {
    return Response.json({ error: "Missing id" }, { status: 400 });
  }

  let result: { code_content: string } | null;
  try {
    result = await env.DB.prepare(
      "SELECT code_content FROM presentations WHERE id = ?1"
    )
      .bind(id)
      .first<{ code_content: string }>();
  } catch (err) {
    console.error("D1 query failed:", err);
    return Response.json({ error: "Database error" }, { status: 500 });
  }

  if (!result) {
    return Response.json({ error: "Not found" }, { status: 404 });
  }

  return Response.json({ code_content: result.code_content });
}

  }, [generatedCode]);

function CodeSyncBack() {
  const { sandpack } = useSandpack();

    const code = sandpack.files[ENTRY_FILE]?.code ?? "";

  }, [sandpack.files]);

  return null;
